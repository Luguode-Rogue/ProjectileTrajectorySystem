﻿// Naval DLC 弹道支持
// 完全反射实现
// 无 DLC 时自动失效
using System;
using System.Collections;
using System.Reflection;
using TaleWorlds.Core;
using TaleWorlds.Engine;
using TaleWorlds.Library;
using TaleWorlds.MountAndBlade;

namespace ProjectileTrajectorySystem
{
    public class NavalDLCmBehavior : MissionLogic
    {
        public override void OnMissionTick(float dt)
        {
            // 只有在启用玩家弹道时才更新，并且至少有一种显示方式被启用
            if (ProjectileTrajectorySettingsManager.Settings.EnableTrajectory &&
                ProjectileTrajectorySettingsManager.Settings.PlayerTrajectory &&
                (ProjectileTrajectorySettingsManager.Settings.UseGameEntityDisplay || ProjectileTrajectorySettingsManager.Settings.UseDebugLineDisplay))
            {
                NavalDLCTrajectorySupport.UpdatePlayerShipTrajectory();
            }
        }
    }
    public static class NavalDLCTrajectorySupport
    {       
        // -----------------------
        // 可配置参数（按需微调）
        // -----------------------
        private const float LOOK_UP_THRESHOLD = 0.2f; // 
        private const float DEFAULT_ALPHA = 1.0f;
        private const float BLUR_ALPHA = 0.03f;          // 目标虚化 alpha（可调）
        private const float ALPHA_EPSILON = 0.01f;      // 小于该值视为无变化
        private const float SMOOTHING_FACTOR_PER_TICK = 0.15f; // 每帧平滑因子（没有 dt 情况下的经验值）
        private static bool _initialized;
        private static bool _navalAvailable;

        private static Type _agentNavalComponentType;
        private static PropertyInfo _steppedShipProperty;
        private static PropertyInfo _shipSiegeWeaponProperty;

        private static FieldInfo _agentComponentsField;

        private static void Init()
        {
            if (_initialized)
                return;

            _initialized = true;

            foreach (var asm in AppDomain.CurrentDomain.GetAssemblies())
            {
                var type = asm.GetType("NavalDLC.Missions.AgentNavalComponent");
                if (type != null)
                {
                    _agentNavalComponentType = type;

                    var missionShipType =
                        asm.GetType("NavalDLC.Missions.Objects.MissionShip");

                    _steppedShipProperty =
                        _agentNavalComponentType.GetProperty("SteppedShip",
                            BindingFlags.Instance | BindingFlags.Public);

                    _shipSiegeWeaponProperty =
                        missionShipType?.GetProperty("ShipSiegeWeapon",
                            BindingFlags.Instance | BindingFlags.Public);

                    _agentComponentsField =
                        typeof(Agent).GetField("_components",
                            BindingFlags.Instance | BindingFlags.NonPublic);

                    _navalAvailable =
                        _steppedShipProperty != null &&
                        _shipSiegeWeaponProperty != null &&
                        _agentComponentsField != null;
                    return;
                }
            }

            _navalAvailable = false;
        }
        public static void UpdatePlayerShipTrajectory()
        {
            Init();

            if (!_navalAvailable)
                return;

            Agent player = Agent.Main;
            if (player == null || !player.IsActive())
                return;

            object navalComponent = GetNavalComponent(player);
            if (navalComponent == null)
                return;

            object missionShip = _steppedShipProperty.GetValue(navalComponent);
            if (missionShip == null)
                return;

            object siegeObj = _shipSiegeWeaponProperty.GetValue(missionShip);
            if (siegeObj == null)
                return;

            if (siegeObj is RangedSiegeWeapon siegeWeapon)
            {
                if (siegeObj != null)
                {
                    bool siegeLookingUp = false;
                    try
                    {
                        // 直接使用引擎提供的方向 
                        Vec3 siegeDir = ProjectileTrajectorySystem.GetSiegeShootingDirection((RangedSiegeWeapon)siegeObj);
                        siegeLookingUp = siegeDir.z > LOOK_UP_THRESHOLD;
                    }
                    catch
                    {
                        siegeLookingUp = false;
                    }

                    // 获取设置：是否启用抬头虚化功能
                    bool enableLookUpBlur = ProjectileTrajectorySettingsManager.Settings.EnableLookUpBlur;

                    // 海战dlc，攻城器抬高时，隐藏攻城器用来显示弹道
                    if (enableLookUpBlur)
                    {
                        SkillSystemBehavior.SetSiegeTargetAlpha((RangedSiegeWeapon)siegeObj, siegeLookingUp ? BLUR_ALPHA : DEFAULT_ALPHA);
                    }
                }
                ProjectileTrajectorySystem.UpdateTrajectory(player, siegeWeapon);
            }
        }

        private static object GetNavalComponent(Agent agent)
        {
            var list = _agentComponentsField.GetValue(agent) as IEnumerable;
            if (list == null)
                return null;

            foreach (var comp in list)
            {
                if (comp == null)
                    continue;

                if (comp.GetType() == _agentNavalComponentType)
                    return comp;
            }

            return null;
        }
    }
}

﻿// ProjectileTrajectorySettingsManager.cs
using System;
using System.IO;
using System.Threading;
using System.Xml.Serialization;
using TaleWorlds.Library;

namespace ProjectileTrajectorySystem
{
    public static class ProjectileTrajectorySettingsManager
    {
        private static readonly string ConfigFolder =
            Path.Combine(BasePath.Name, "Modules", "ProjectileTrajectorySystem");

        private static readonly string XmlPath =
            Path.Combine(ConfigFolder, "ProjectileTrajectorySettings.xml");

        private static readonly object _lock = new object();
        private static FileSystemWatcher _watcher;
        private static bool _reloadRequested;
        private static DateTime _lastReloadTime = DateTime.MinValue;

        public static ProjectileTrajectorySettingsData Data { get; private set; }

        public static ProjectileTrajectorySettingsData Settings
        {
            get
            {
                if (Data == null)
                    Load();
                return Data;
            }
        }

        public static void Load()
        {
            lock (_lock)
            {
                if (!Directory.Exists(ConfigFolder))
                    Directory.CreateDirectory(ConfigFolder);

                if (File.Exists(XmlPath))
                {
                    LoadFromXml();
                }
                else
                {
                    Data = new ProjectileTrajectorySettingsData();
                    SaveToXml();
                }

                StartWatcher();
            }
        }

        public static void Save()
        {
            lock (_lock)
            {
                SaveToXml();
            }
        }

        private static void LoadFromXml()
        {
            try
            {
                var serializer = new XmlSerializer(typeof(ProjectileTrajectorySettingsData));
                using var stream = new FileStream(XmlPath, FileMode.Open, FileAccess.Read, FileShare.ReadWrite);
                Data = (ProjectileTrajectorySettingsData)serializer.Deserialize(stream)
                       ?? new ProjectileTrajectorySettingsData();
            }
            catch
            {
                Data = new ProjectileTrajectorySettingsData();
            }
        }

        private static void SaveToXml()
        {
            try
            {
                var serializer = new XmlSerializer(typeof(ProjectileTrajectorySettingsData));
                using var stream = new FileStream(XmlPath, FileMode.Create, FileAccess.Write, FileShare.Read);
                serializer.Serialize(stream, Data ?? new ProjectileTrajectorySettingsData());
            }
            catch { }
        }

        private static void StartWatcher()
        {
            if (_watcher != null)
                return;

            _watcher = new FileSystemWatcher(ConfigFolder, "ProjectileTrajectorySettings.xml");
            _watcher.NotifyFilter = NotifyFilters.LastWrite | NotifyFilters.Size;
            _watcher.Changed += OnXmlChanged;
            _watcher.EnableRaisingEvents = true;
        }

        private static void OnXmlChanged(object sender, FileSystemEventArgs e)
        {
            // 防止连续触发
            var now = DateTime.Now;
            if ((now - _lastReloadTime).TotalMilliseconds < 300)
                return;

            _lastReloadTime = now;
            _reloadRequested = true;
        }

        /// <summary>
        /// 在主线程（如 Mission Tick）中调用
        /// </summary>
        public static void ProcessPendingReload()
        {
            if (!_reloadRequested)
                return;

            lock (_lock)
            {
                _reloadRequested = false;
                LoadFromXml();

                InformationManager.DisplayMessage(
                    new InformationMessage("ProjectileTrajectorySystem: XML 热重载完成"));
            }
        }

        public static void SyncFromMCM(ProjectileTrajectorySettings mcm)
        {
            if (mcm == null) return;
            lock (_lock)
            {
                if (Data == null) Data = new ProjectileTrajectorySettingsData();

                Data.EnableTrajectory = mcm.EnableTrajectory;
                Data.PlayerTrajectory = mcm.PlayerTrajectory;
                Data.EnemyTrajectory = mcm.EnemyTrajectory;
                Data.MissileTrajectory = mcm.MissileTrajectory;
                Data.EnemyHighlight = mcm.EnemyHighlight;
                Data.UseGameEntityDisplay = mcm.UseGameEntityDisplay;
                Data.UseDebugLineDisplay = mcm.UseDebugLineDisplay;
                Data.UseGameEntityForPlayerRanged = mcm.UseGameEntityForPlayerRanged;
                Data.UseGameEntityForPlayerSiege = mcm.UseGameEntityForPlayerSiege;
                Data.MaxTrackedEnemiesLegacy = mcm.MaxTrackedEnemiesLegacy;
                Data.EnableSlowMotion = mcm.EnableSlowMotion;
                Data.SlowMoEnemyRadius = mcm.SlowMoEnemyRadius;
                Data.SlowMoTimeScale = mcm.SlowMoTimeScale;
                Data.EnableLeadPrediction = mcm.EnableLeadPrediction;

                SaveToXml();
            }
        }
    }
}

﻿// 统一的投射物弹道计算与可视化系统
// 支持：
// - 玩家远程武器
// - 攻城远程武器
// - 敌方 AI 预测弹道 (旧版)
// - 新版：基于 OnAgentShootMissile 的一次性弹道计算与显示
// - GameEntity 和 DebugLine 两种显示方式，可单独或同时启用
using System;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using TaleWorlds.Core;
using TaleWorlds.Engine;
using TaleWorlds.Library;
using TaleWorlds.MountAndBlade;
using static TaleWorlds.Library.MathF;

namespace ProjectileTrajectorySystem
{
    public static class ProjectileTrajectorySystem
    {
        #region ===== 内部状态 =====

        private static bool _hasCollided;
        private static Vec3 _collisionPoint = Vec3.Invalid;
        private static bool _isFirstPoint;
        private static Vec3 _lastPoint;
        private static float _airFriction;

        #endregion

        #region ===== 玩家 / 攻城武器弹道 =====

        public static void UpdateTrajectory(Agent agent, RangedSiegeWeapon siegeWeapon = null)
        {
            if (agent == null || !agent.IsActive()) return;

            if (siegeWeapon == null)
            {
                // 没有攻城武器，不执行任何操作，保留之前的轨迹（例如玩家远程武器轨迹）
                return;
            }

            // --- 修改：检查细分设置 ---
            bool shouldShowPlayerSiegeGE = ProjectileTrajectorySettingsManager.Settings.UseGameEntityDisplay &&
                                           ProjectileTrajectorySettingsManager.Settings.UseGameEntityForPlayerSiege;
            bool shouldShowPlayerSiegeDL = ProjectileTrajectorySettingsManager.Settings.UseDebugLineDisplay;

            // 如果 GameEntity 和 DebugLine 都不显示攻城轨迹，则直接返回
            if (!shouldShowPlayerSiegeGE && !shouldShowPlayerSiegeDL)
            {
                return;
            }

            // 如果需要显示 GameEntity，才清理旧轨迹
            if (shouldShowPlayerSiegeGE)
            {
                ClearNormalTrajectory(); // 清理旧轨迹，为新轨迹腾出空间
            }

            ResetFrameState();

            float speed = GetSiegeShootingSpeed(siegeWeapon);
            Vec3 direction = GetSiegeShootingDirection(siegeWeapon);
            Vec3 startPos = GetSiegeProjectileStartPosition(siegeWeapon);

            // 调用 SimulateTrajectory，它会根据 ProjectileTrajectorySettingsManager.Settings 的全局设置进行绘制
            SimulateTrajectory(
                startPos,
                direction,
                speed,
                0.3f,
                4.8f,
                0.15f,
                timeKeyOffset: 0f
            // customLine 默认为 null，意味着使用 WoW_Line
            // color 默认为 0xFF00FFFFu
            ); 

        }

        public static void UpdateTrajectoryRangeWeapon(Agent agent)
        {
            if (agent == null || !agent.IsActive()) return;

            bool shouldShowPlayerRangedGE = false;
            EquipmentIndex weaponIndex = agent.GetPrimaryWieldedItemIndex();
            if (weaponIndex == EquipmentIndex.None)
            {
                shouldShowPlayerRangedGE = ProjectileTrajectorySettingsManager.Settings.UseGameEntityDisplay &&
                                           ProjectileTrajectorySettingsManager.Settings.UseGameEntityForPlayerRanged;
                if (shouldShowPlayerRangedGE && SkillSystemBehavior.WoW_Line.Count > 0)
                {
                    ClearNormalTrajectory();
                }
                return;
            }

            MissionWeapon weapon = agent.Equipment[weaponIndex];
            if (weapon.IsEmpty || !weapon.CurrentUsageItem.IsRangedWeapon)
            {
                shouldShowPlayerRangedGE = ProjectileTrajectorySettingsManager.Settings.UseGameEntityDisplay &&
                                           ProjectileTrajectorySettingsManager.Settings.UseGameEntityForPlayerRanged;
                if (shouldShowPlayerRangedGE && SkillSystemBehavior.WoW_Line.Count > 0)
                {
                    ClearNormalTrajectory();
                }
                return;
            }

            shouldShowPlayerRangedGE = ProjectileTrajectorySettingsManager.Settings.UseGameEntityDisplay &&
                                       ProjectileTrajectorySettingsManager.Settings.UseGameEntityForPlayerRanged;
            bool shouldShowPlayerRangedDL = ProjectileTrajectorySettingsManager.Settings.UseDebugLineDisplay;

            if (!shouldShowPlayerRangedGE && !shouldShowPlayerRangedDL)
            {
                return;
            }

            if (shouldShowPlayerRangedGE)
            {
                ClearNormalTrajectory();
            }

            ResetFrameState();

            // === 新增：优先使用缓存速度 ===
            float speed;
            var ammoItem = weapon.AmmoWeapon;
            string weaponId = weapon.Item?.Name?.ToString() ?? string.Empty;
            string ammoId = ammoItem.Item?.Name?.ToString() ?? string.Empty;
            if (ammoId == null || ammoId.Equals(""))
            {
                ammoId = weaponId;
            }
            var key = (agent.Index, weaponId, ammoId);

            if (SkillSystemBehavior.ProjectileSpeedCache.Count>0 &&
                SkillSystemBehavior.ProjectileSpeedCache.TryGetValue(key, out float cachedSpeed) == true)
            {
                speed = cachedSpeed;
            }

        private static void SimulateTrajectory(
    Vec3 startPos,
    Vec3 direction,
    float speed,
    float timeStart,
    float timeEnd,
    float timeStep,
    float timeKeyOffset,
    Dictionary<float, GameEntity> customLine = null,
    uint baseColor = 0xFF00FFFFu)
        {
            // 检查是否至少有一种显示方式开启
            if (!ProjectileTrajectorySettingsManager.Settings.UseGameEntityDisplay && !ProjectileTrajectorySettingsManager.Settings.UseDebugLineDisplay)
            {
                // 如果两种显示方式都关闭，则直接返回，不进行任何计算或绘制
                return;
            }

            direction.Normalize();
            Vec3 previousPos = startPos; // 初始化上一个点为起点

            // 预设颜色数组，用于清晰区分不同距离的线段
            uint[] colorArray = {
        0xFF00FFFFu, // 青色 - 起始
        0xFF0000FFu, // 红色
        0xFF00FF00u, // 绿色
        0xFFFF0000u, // 蓝色
        0xFF8000FFu, // 紫色
        0xFFFFFF00u, // 黄色
        0xFFFF00FFu, // 品红
        0xFF00FFFFu, // 青色
        0xFFFFA500u, // 橙色
        0xFF808080u, // 灰色
        0xFF000080u, // 深蓝
        0xFF008000u, // 深绿
        0xFF800000u, // 深红
        0xFF800080u, // 深紫
        0xFF008080u, // 深青
        0xFF808000u  // 深黄
    };

            int colorIndex = 0;

            for (float t = timeStart; t <= timeEnd; t += timeStep)
            {
                if (_hasCollided) break;

                Vec3 currentPos = CalculatePosition(startPos, direction, speed, t);

                // 根据设置选择显示方式
                if (ProjectileTrajectorySettingsManager.Settings.UseGameEntityDisplay)
                {
                    // 使用 GameEntity 显示 - 传递原始的 customLine
                    // 如果 customLine 是 null，PlaceGameEntityMarker 会使用 WoW_Line
                    PlaceGameEntityMarker(currentPos, t + timeKeyOffset, customLine, baseColor);
                }

                if (ProjectileTrajectorySettingsManager.Settings.UseDebugLineDisplay)
                {
                    // 使用预设颜色数组中的颜色，循环使用
                    uint segmentColor = colorArray[colorIndex % colorArray.Length];

                    // 使用 DebugLine 显示线段，每个线段有明显的不同颜色
                    DrawDebugLineSegment(previousPos, currentPos - previousPos, segmentColor, 0f);

                    // 更新颜色索引
                    colorIndex++;
                }

                previousPos = currentPos; // 更新上一个点
            }
        }
        // --- 新增：用于在 OnMissionTick 中持续绘制 DebugLine ---
        public static void DrawDebugLineTrajectory(List<Vec3> points, uint color)
        {
            if (points == null || points.Count < 2) return; // Need at least 2 points to draw a line

            for (int i = 1; i < points.Count; i++)
            {
                Vec3 previousPos = points[i - 1];
                Vec3 currentPos = points[i];
                Vec3 direction = currentPos - previousPos;
                RenderDebugLine(previousPos, direction, color, true, 0f); // Time 0 means it's drawn this frame only
            }
        }

        private static Vec3 CalculatePosition(Vec3 origin, Vec3 direction, float speed, float time, float airFrictionOverride = -1f)
        {
            const float dt = 0.01f;
            Vec3 velocity = direction * speed;
            Vec3 position = origin;

            float frictionToUse = airFrictionOverride >= 0 ? airFrictionOverride : _airFriction;

            for (float t = 0; t < time; t += dt)
            {
                float v = velocity.Length;
                if (v > 0.001f)
                    velocity -= velocity.NormalizedCopy() * (frictionToUse * v * v * dt);
                velocity.z -= 9.81f * dt;
                position += velocity * dt;
            }

            return position;
        }

        // HandleCollision 保持不变，但只应用于 WoW_Line 的 GameEntity 清理
        private static void HandleCollision(Vec3 pos, float timeKey)
        {
            if (_hasCollided) return;

            if (Mission.Current.Scene.RayCastForClosestEntityOrTerrain(
                    pos,
                    pos + Vec3.Up * 5000f,
                    out float dist,
                    out Vec3 hitPoint,
                    out _,
                    1f) && dist < 5f)
            {
                _hasCollided = true;
                _collisionPoint = hitPoint;
                // GenerateImpactCircle 也应只应用于 WoW_Line
                if (ProjectileTrajectorySettingsManager.Settings.UseGameEntityDisplay)
                {
                    GenerateImpactCircle(hitPoint);
                }
                // ClearTrajectoryAfterCollision 也只应用于 WoW_Line
                if (ProjectileTrajectorySettingsManager.Settings.UseGameEntityDisplay)
                {
                    ClearTrajectoryAfterCollision(timeKey);
                }
            }
        }

        private static void PlaceGameEntityMarker(
            Vec3 pos,
            float timeKey,
            Dictionary<float, GameEntity> line,
            uint color)
        {
            line ??= SkillSystemBehavior.WoW_Line;

            if (!line.TryGetValue(timeKey, out var entity))
            {
                // 从池中获取或创建新实体
                entity = SkillSystemBehavior.GetOrCreateGameEntity();
                // *不* 立即设置颜色，因为新获取的实体颜色缓存已被清除
                // entity.SetContourColor(color, true); // 移除这行

                // 注意：模型实例化可能需要优化，如果每次都执行可能仍有性能开销。
                // 但目前保持与原逻辑一致，如果模型固定，可以考虑预设或缓存。
                entity.AddAllMeshesOfGameEntity(
                    GameEntity.Instantiate(
                        Mission.Current.Scene,
                        "mangonel_mapicon_projectile",
                        true));
                line[timeKey] = entity;
            }

            entity.SetLocalPosition(pos);

            // --- 新增：性能优化 + 遮挡检测：检查颜色缓存并决定是否设置描边 ---
            // 1. 获取玩家位置
            Agent player = Agent.Main;
            if (player == null || !player.IsActive())
            {
                // 如果玩家不存在或无效，暂时不清除颜色缓存，但不设置描边
                // 可以选择在这里清除描边，但通常这种情况很少见
                return;
            }

            Vec3 playerEyePos = player.GetEyeGlobalPosition();

            // 2. 执行遮挡检测
            float distanceToEntity = (pos - playerEyePos).Length;
            bool isOccluded = Mission.Current.Scene.RayCastForClosestEntityOrTerrain(
                playerEyePos,
                pos,
                out float hitDistance,
                out Vec3 hitPoint,
                out _,
                0.1f);

            bool isVisible = !(isOccluded && hitDistance < distanceToEntity);

            // 3. 根据可见性决定描边颜色
            uint effectiveColor = isVisible ? color : 0x00000000u; // 如果不可见，使用透明色 (0) 或 null，这里用 0x00000000u 便于缓存比较

            // 4. 检查缓存并设置描边（性能优化）
            if (!SkillSystemBehavior.GameEntityColorCache.TryGetValue(entity, out var cachedColor) || cachedColor != effectiveColor)
            {
                // 缓存中没有记录，或者颜色不同，则更新颜色并缓存
                // 如果 effectiveColor 是透明色 (0x00000000u)，SetContourColor(null, true) 会移除描边
                entity.SetContourColor(isVisible ? color : null, true);
                SkillSystemBehavior.GameEntityColorCache[entity] = effectiveColor;
            }
            // 如果颜色相同（无论是可见的 color 还是不可见的 0x00000000u），则跳过 SetContourColor 调用
            // --- 结束新增 ---
        }
        private static void ClearTrajectoryAfterCollision(float collisionKey)
        {
            foreach (var kv in SkillSystemBehavior.WoW_Line.ToList())
            {
                if (kv.Key > collisionKey && kv.Key < 100f)
                {
                    // 归还 GameEntity 到池中
                    SkillSystemBehavior.ReturnGameEntityToPool(kv.Value);
                    SkillSystemBehavior.WoW_Line.Remove(kv.Key); // 从字典中移除
                }
            }
        }

        private static void GenerateImpactCircle(Vec3 center)
        {
            const int count = 8;
            const float radius = 2f;

            for (int i = 0; i < count; i++)
            {
                float key = 100 + i;
                float angle = TaleWorlds.Library.MathF.PI * 2f * i / count;
                Vec3 pos = new Vec3(
                    center.x + TaleWorlds.Library.MathF.Cos(angle) * radius,
                    center.y + TaleWorlds.Library.MathF.Sin(angle) * radius,
                    center.z);

                PlaceGameEntityMarker(pos, key, SkillSystemBehavior.WoW_Line, 0xFF00FFFFu);
            }
        }

        public static Vec3 GetSiegeProjectileStartPosition(RangedSiegeWeapon weapon)
        {
            if (weapon == null) return Vec3.Invalid;

            // 检查是否是 Ballista 类型（包括 ShipBallista 等派生类）
            if (weapon is Ballista ballistaInstance)
            {
                // 对于 Ballista，直接使用 ProjectileEntityCurrentGlobalPosition
                // 这是引擎内部在发射时设置的精确发射点位置
                return ballistaInstance.ProjectileEntityCurrentGlobalPosition;
            }
            // 如果不是 Ballista 类型，则沿用原有的逻辑
            // 这适用于 Mangonel, Trebuchet 等使用 "clean"/"projectile_leaving_position" 的武器
            else
            {
                WeakGameEntity weaponEntity = weapon.GameEntity;
                if (weaponEntity == null) return Vec3.Invalid;

                // 1️⃣ 找到 "clean" 子节点
                WeakGameEntity clean = WeakGameEntity.Invalid;
                var children = weaponEntity.GetChildren();
                if (children != null)
                {
                    clean = children.FirstOrDefault(x => x != null && x.Name == "clean");
                }

                if (clean == null) return Vec3.Invalid;

                // 2️⃣ 从 WeakGameEntity 转换成 GameEntity
                GameEntity launcher = GameEntity.CreateFromWeakEntity(clean);
                if (launcher == null) return Vec3.Invalid;

                // 3️⃣ 查找 projectile_leaving_position 节点
                var launcherChildren = launcher.GetChildren();
                if (launcherChildren == null) return Vec3.Invalid;

                var projectileNode = launcherChildren.FirstOrDefault(x => x != null && (x.Name == "projectile_leaving_position" || x.Name == "use_pos"));
                if (projectileNode == null) return Vec3.Invalid;

                // 4️⃣ 返回坐标
                Vec3 position = projectileNode.GlobalPosition;
                return position;
            }
        }
        public static float GetWeaponAirFriction(Agent agent, out EquipmentIndex _)
        {
            _ = EquipmentIndex.None;

            // 检查是否是人类角色
            if (!agent.IsHuman) return 0f;

            // 获取当前武器
            var weapon = agent.WieldedWeapon;
            if (weapon.IsEmpty) return 0f;

            // 获取武器的ItemObject
            var item = weapon.Item;
            if (item == null || item.WeaponComponent == null)
            {
                return 0f;
            }

            // 获取PrimaryWeapon的WeaponClass和WeaponFlags
            var primaryWeapon = item.WeaponComponent.PrimaryWeapon;
            if (primaryWeapon == null)
            {
                return 0f;
            }

            // 使用引擎提供的标准方法获取空气阻力
            float outAirFrictionConstant = ItemObject.GetAirFrictionConstant(primaryWeapon.WeaponClass, primaryWeapon.WeaponFlags);
            //InformationManager.DisplayMessage(new InformationMessage(outAirFrictionConstant.ToString()));
            return outAirFrictionConstant;
        }

        private static List<Vec3> SimulateTrajectoryForMissile(Vec3 startPos, Vec3 direction, float speed, float airFriction, float timeStep, float maxTime, float timeKeyOffset, Dictionary<float, GameEntity> missileLine, uint color)
        {
            // 检查是否至少有一种显示方式开启
            if (!ProjectileTrajectorySettingsManager.Settings.UseGameEntityDisplay && !ProjectileTrajectorySettingsManager.Settings.UseDebugLineDisplay)
            {
                // 如果两种显示方式都关闭，则直接返回，不进行任何计算或绘制
                return null; // 返回 null 表示没有计算轨迹点
            }

            direction.Normalize();
            float originalAirFriction = _airFriction; // 保存原始值
            _airFriction = airFriction; // 设置当前计算使用的空气阻力

            Vec3 previousPos = startPos; // 初始化上一个点为起点
            List<Vec3> calculatedPoints = new List<Vec3>(); // 存储计算出的点

            for (float t = 0f; t <= maxTime; t += timeStep)
            {
                Vec3 currentPos = CalculatePosition(startPos, direction, speed, t, airFriction);
                calculatedPoints.Add(currentPos); // 添加到列表

                // 根据设置选择显示方式
                if (ProjectileTrajectorySettingsManager.Settings.UseGameEntityDisplay)
                {
                    // 使用 GameEntity 显示
                    PlaceGameEntityMarker(currentPos, t + timeKeyOffset, missileLine, color);
                }

                if (ProjectileTrajectorySettingsManager.Settings.UseDebugLineDisplay)
                {
                    // 使用 DebugLine 显示线段 (即时绘制)
                    DrawDebugLineSegment(previousPos, currentPos - previousPos, color, 0f);
                }

                previousPos = currentPos; // 更新上一个点
            }

            _airFriction = originalAirFriction; // 恢复原始值

            return calculatedPoints; // 返回计算好的点列表
        }

        // 在 ProjectileTrajectorySystem 类中添加
        public static Vec3 CalculateLeadPosition(Agent player, Agent target, float projectileSpeed)
        {
            Vec3 shooterPos = player.GetEyeGlobalPosition();
            Vec3 targetCurrentPos = target.GetEyeGlobalPosition();

            // 1. 获取目标速度 (需要 SkillSystemBehavior 中维护的数据)
            Vec3 targetVelocity = Vec3.Zero;
            if (SkillSystemBehavior.TrackedAgents.TryGetValue(target, out var data))
            {
                targetVelocity = data.Velocity;
            }

            // 2. 初始时间估算 (距离 / 速度)
            float horizontalDistance = (targetCurrentPos.AsVec2 - shooterPos.AsVec2).Length;
            float timeToTarget = horizontalDistance / projectileSpeed;

            // 3. 迭代拟合 (核心算法)
            // 原理：因为要打提前量，枪口抬起导致弹道变长 -> 飞行时间变长 -> 需要更多提前量
            float previousTime = 0;
            float currentTime = timeToTarget;
            int iterations = 0;
            const int maxIterations = 5;

            while (TaleWorlds.Library.MathF.Abs(previousTime - currentTime) > 0.001f && iterations < maxIterations)
            {
                previousTime = currentTime;

                // 预测目标的新位置 = 当前位置 + 速度 * 预估时间
                Vec3 predictedTargetPos = targetCurrentPos + (targetVelocity * previousTime);

                // 重新计算弹道飞行时间
                // 这里简化了，实际应考虑抛物线。为了精确，我们调用抛物线解算函数
                Vec3 shotDir = CalculateProjectileFiringSolution(shooterPos, predictedTargetPos, projectileSpeed, 9.81f);

                if (shotDir!=Vec3.Invalid && shotDir!=Vec3.Zero)
                {
                    // 如果无解（比如距离太远），回退到直线计算
                    Vec3 simpleDelta = predictedTargetPos - shooterPos;
                    currentTime = simpleDelta.AsVec2.Length / projectileSpeed;
                }
                else
                {
                    // 这里需要一个更精确的飞行时间估算
                    // 作为简化，我们使用直线距离除以速度分量
                    // (注：如果要极致精确，需要积分计算抛物线飞行时间，这里先用简化版)
                    Vec3 velocityVector = shotDir * projectileSpeed;
                    float airTime = (velocityVector.z + TaleWorlds.Library.MathF.Sqrt(velocityVector.z * velocityVector.z + 2 * 9.81f * (shooterPos.z - predictedTargetPos.z))) / 9.81f;
                    currentTime = airTime;
                }
            }

            // 4. 返回最终预测点
            return targetCurrentPos + (targetVelocity * currentTime);
        }

        // 复用你提供的抛物线解算函数 (放在 ProjectileTrajectorySystem 类中)
        public static Vec3 CalculateProjectileFiringSolution(Vec3 start, Vec3 end, float speed, float gravity)
        {
            Vec2 horizontalDistance = new Vec2(end.x - start.x, end.y - start.y);
            float horizontalRange = horizontalDistance.Length;
            float verticalDistance = end.z - start.z;

            float speedSquared = speed * speed;
            float sqrtTerm = speedSquared * speedSquared - gravity * (gravity * horizontalRange * horizontalRange + 2 * verticalDistance * speedSquared);

            if (sqrtTerm < 0.0f)
            {
                return Vec3.Invalid;
            }

            float sqrtValue = (float)Math.Sqrt(sqrtTerm);
            float angle = (float)Math.Atan2(speedSquared - sqrtValue, gravity * horizontalRange);

            Vec3 firingSolution = new Vec3(horizontalDistance.x, horizontalDistance.y, 0);
            firingSolution.Normalize();
            firingSolution *= (float)Math.Cos(angle) * speed;
            firingSolution.z = (float)Math.Sin(angle) * speed;

            // 注意：这里不要 Normalize()，保留速度矢量用于计算飞行时间
            // 如果返回的是方向，才需要 Normalize。根据你的调用逻辑决定。
            // firingSolution.Normalize(); 
            return firingSolution;
        }
    }

        private static readonly Dictionary<RangedSiegeWeapon, List<WeakGameEntity>> _siegeChildEntities = new();
        public static void SetSiegeTargetAlpha(RangedSiegeWeapon siege, float targetAlpha)
        {
            if (siege == null) return;
            try
            {
                // 第一次调用时收集所有子实体
                if (!_siegeChildEntities.ContainsKey(siege))
                {
                    var entities = new List<WeakGameEntity>();

                    WeakGameEntity highestParent = GetHighestParent(siege.GameEntity);
                    if (highestParent != null)
                    {

                    }

                }

                foreach (var entity in _siegeChildEntities[siege])
                {

                        if (entity != null && !WeakGameEntity.Invalid.Equals(entity))

                    }

                }
            }

        // 辅助方法 - 递归收集子实体
        private static void CollectChildren(WeakGameEntity parent, List<WeakGameEntity> children)
        {
            if (parent == null || WeakGameEntity.Invalid.Equals(parent)) return;

            try
            {
                var ge = GameEntity.CreateFromWeakEntity(parent);
                if (ge == null) return;

                // 先添加当前父实体（如果还没添加）
                if (!children.Contains(parent))
                {
                    children.Add(parent);
                }

                foreach (var child in ge.GetChildren())
                {
                    if (ge != null)
                    {
                        var weak = child.WeakEntity;
                        if (!children.Contains(weak))
                        {
                            children.Add(weak);
                            CollectChildren(weak, children);
                        }
                    }
                }
            }
            catch { }
        }

        // 新增方法：获取最高父实体
        private static WeakGameEntity GetHighestParent(WeakGameEntity entity)
        {
            if (entity == null || WeakGameEntity.Invalid.Equals(entity))
                return WeakGameEntity.Invalid;

            // 递归找到最高父实体
            while (entity.Parent != null && !WeakGameEntity.Invalid.Equals(entity.Parent))
            {
                entity = entity.Parent;
            }

            return entity;
        }

        // -----------------------
        // 平滑并应用 alpha 到实体（每帧调用）
        // -----------------------
        private static void UpdateAlphaSmoothing()
        {
            // 采用快照键集合，防止在迭代中修改字典导致异常
            var ids = new List<int>(_targetAlphaById.Keys);
            foreach (int id in ids)
            {
                // 若实体不存在则清理缓存
                if (!_entityById.TryGetValue(id, out var ge) || ge == null)
                {
                    _targetAlphaById.Remove(id);
                    _currentAlphaById.Remove(id);
                    _entityById.Remove(id);
                    continue;
                }

                float target = _targetAlphaById.ContainsKey(id) ? _targetAlphaById[id] : DEFAULT_ALPHA;
                float current = _currentAlphaById.ContainsKey(id) ? _currentAlphaById[id] : DEFAULT_ALPHA;

                float factor = SMOOTHING_FACTOR_PER_TICK; // 如有 dt，可以改为 MathF.Clamp(dt * speed, 0, 1)

                float next = current + (target - current) * factor;

                // 只有出现明显变化才调用 SetAlpha
                if (Math.Abs(next - current) > ALPHA_EPSILON)
                {
                    try
                    {
                        // 防守性检查
                        if (!ge.IsGhostObject())
                        {
                            ge.SetAlpha(next);
                        }
                    }
                    catch
                    {
                        // 引擎版本差异导致失败时忽略，避免崩溃
                    }
                    _currentAlphaById[id] = next;
                }
                else
                {
                    // 变化很小，认为已经到达目标
                    _currentAlphaById[id] = target;

                    // 若目标为默认 alpha，则清理缓存，减少维护开销
                    if (Math.Abs(target - DEFAULT_ALPHA) <= ALPHA_EPSILON)
                    {
                        _targetAlphaById.Remove(id);
                        _currentAlphaById.Remove(id);
                        _entityById.Remove(id);
                    }
                }
            }
        }

        // -----------------------
        // 清理：恢复所有被管理实体的 alpha（在 Mission 结束或卸载时调用）
        // -----------------------
        private static void RestoreAllAlphas()
        {
            var ids = new List<int>(_entityById.Keys);
            foreach (var id in ids)
            {
                if (_entityById.TryGetValue(id, out var ge) && ge != null)
                {
                    try { ge.SetAlpha(DEFAULT_ALPHA); } catch { }
                }
                _entityById.Remove(id);
                _currentAlphaById.Remove(id);
                _targetAlphaById.Remove(id);
            }
        }

        public override void OnAgentShootMissile(Agent shooterAgent, EquipmentIndex weaponIndex, Vec3 position, Vec3 velocity, Mat3 orientation, bool hasRigidBody, int forcedMissileIndex)
        {
            base.OnAgentShootMissile(shooterAgent, weaponIndex, position, velocity, orientation, hasRigidBody, forcedMissileIndex);
            debugString2 =
                shooterAgent.GetEyeGlobalPosition().X.ToString() + " " +
                shooterAgent.GetEyeGlobalPosition().Y.ToString() + " " +
                shooterAgent.GetEyeGlobalPosition().Z.ToString() + " " +
                velocity.Length.ToString();
            if (shooterAgent == Agent.Main)
            {
                InformationManager.DisplayMessage(new InformationMessage(debugString1));
                InformationManager.DisplayMessage(new InformationMessage(debugString2));
            }
            // 检查设置是否启用投射物弹道
            if (!ProjectileTrajectorySettingsManager.Settings.EnableTrajectory || !ProjectileTrajectorySettingsManager.Settings.MissileTrajectory)
                return;

            // 获取玩家代理对象
            Agent playerAgent = Agent.Main;
            if (playerAgent == null) return; // 防御性检查，确保玩家存在

            // 判断是否是玩家发射或者敌人以玩家为目标的情况
            bool isPlayerProjectile = shooterAgent == playerAgent;
            bool isEnemyTargetingPlayer = shooterAgent != playerAgent
                                           && shooterAgent.Team != null && playerAgent.Team != null // 检查 Team 是否为 null
                                           && shooterAgent.Team.IsEnemyOf(playerAgent.Team)
                                           && shooterAgent.GetTargetAgent() == playerAgent; // 检查 Target 是否为 null (GetTargetAgent 可能返回 null)

            // 如果都不是，则不处理此投射物的轨迹
            if (!isPlayerProjectile && !isEnemyTargetingPlayer)
            {
                return; // 直接返回，不执行后续的轨迹计算和显示
            }

            // 防御性检查
            if (shooterAgent == null || !shooterAgent.IsActive())
                return;


            // 获取新生成的 Missile (假设列表末尾就是最新的)
            var missilesList = Mission.Current.MissilesList;
            if (missilesList.Count == 0) return; // 防御性检查

            var newMissile = missilesList[missilesList.Count - 1];
            if (newMissile == null) return; // 防御性检查

            // 获取计算所需参数
            var missileIndex = newMissile.Index;
            var startPos = position; // 使用 OnAgentShootMissile 提供的 position
            var speed = velocity.Length;
            var direction = speed > 0 ? velocity.NormalizedCopy() : orientation.f; // 使用 velocity 方向，或 fallback 到 orientation
            var airFriction = ProjectileTrajectorySystem.GetWeaponAirFriction(shooterAgent, out _); // 尝试获取发射者武器的空气阻力


            // 获取武器和弹药
            var weapon = shooterAgent.Equipment[weaponIndex];
            if (!weapon.IsEmpty)
            {
                var weaponItem = weapon.Item;
                var ammoItem = newMissile.Weapon.Item; // 对于弓/弩，通常是箭矢；对于投掷武器，可能是自身

                int agentIndex = shooterAgent.Index;
                String weaponId = weaponItem?.Name.ToString();
                String ammoId = ammoItem?.Name.ToString();

                var key = (agentIndex, weaponId, ammoId);

                // 缓存初速（只记录首次，或可选择 always update）
                if (!ProjectileSpeedCache.ContainsKey(key))
                {
                    ProjectileSpeedCache[key] = speed;
                }
            }
            // 一次性计算并更新轨迹显示
            ProjectileTrajectorySystem.UpdateMissileTrajectory(missileIndex, startPos, direction, speed, airFriction);
        }

        private void UpdateEnemyTracking()
        {
            Agent player = Agent.Main;
            if (player == null) return;

            // 清理失效
            for (int i = _trackedEnemies.Count - 1; i >= 0; i--)
            {
                if (!IsEnemyAimingPlayer(_trackedEnemies[i], player))
                {
                    ProjectileTrajectorySystem.ClearEnemyTrajectory(_trackedEnemies[i].Index);
                    _trackedEnemies.RemoveAt(i);
                }
            }

            if (_trackedEnemies.Count >= MaxTrackedEnemies) return;

            foreach (Agent agent in Mission.Current.Agents)
            {
                if (_trackedEnemies.Count >= MaxTrackedEnemies) break;
                if (_trackedEnemies.Contains(agent)) continue;
                if (IsEnemyAimingPlayer(agent, player))
                    _trackedEnemies.Add(agent);
            }
        }

        private void UpdateEnemyTrajectories()
        {
            foreach (Agent enemy in _trackedEnemies)
                ProjectileTrajectorySystem.UpdateEnemyTrajectory(enemy);
        }

        private static bool IsEnemyAimingPlayer(Agent agent, Agent player)
        {
            if (agent == null || player == null) return false;
            if (!agent.IsActive()) return false;

            // Team 判空（非常关键）
            if (agent.Team == null || player.Team == null || !agent.Team.IsValid || !player.Team.IsValid) return false;
            if (!agent.Team.IsEnemyOf(player.Team)) return false;

            // 目标判空
            Agent target = agent.GetTargetAgent();
            if (target == null || target != player) return false;

            // 武器判空
            MissionWeapon weapon = agent.WieldedWeapon;
            if (weapon.IsEmpty) return false;
            if (!weapon.CurrentUsageItem.IsRangedWeapon) return false;

            // 上半身动画：拉弓 / 瞄准阶段
            return agent.GetCurrentActionStage(1) == Agent.ActionStage.AttackReady;
        }

        private static void UpdatePlayerTrajectory()
        {
            Agent player = Agent.Main;
            if (player == null) return;

            // 获取玩家当前武器
            MissionWeapon weapon = player.WieldedWeapon;
            bool hasValidWeapon = !weapon.IsEmpty;
            bool isUsingRangedWeapon = hasValidWeapon && weapon.CurrentUsageItem.IsRangedWeapon;

            // 获取设置：是否启用攻击准备阶段限制
            bool enableAttackReadyRestriction = ProjectileTrajectorySettingsManager.Settings.EnableAttackReadyRestriction;
            bool isAttacking = player.GetCurrentActionStage(1) == Agent.ActionStage.AttackReady;

            // 根据设置决定是否需要攻击准备阶段的限制
            bool shouldUpdateTrajectory = enableAttackReadyRestriction ? (isUsingRangedWeapon && isAttacking) : isUsingRangedWeapon;

            // 1) 使用 agent.LookDirection 判断玩家是否"抬头"
            // 获取设置：是否启用抬头虚化功能
            bool enableLookUpBlur = ProjectileTrajectorySettingsManager.Settings.EnableLookUpBlur;
            Vec3 lookDir = player.LookDirection;
            bool playerLookingUp = lookDir.z > LOOK_UP_THRESHOLD && isUsingRangedWeapon && enableLookUpBlur;

            // 2) 为玩家设置目标 alpha（抬头时虚化，根据设置决定）
            if (enableLookUpBlur)
            {
                SetAgentTargetAlpha(player, playerLookingUp ? BLUR_ALPHA : DEFAULT_ALPHA);
            }

            // --- 分支1: 远程武器逻辑 ---
            if (shouldUpdateTrajectory)
            {
                // 检查玩家是否处于远程武器的攻击动作中
                // 根据设置决定是否仅在攻击准备阶段更新轨迹

                // 3) 调用远程武器轨迹
                try
                {
                    ProjectileTrajectorySystem.UpdateTrajectoryRangeWeapon(player);
                }
                catch
                {
                    // 守护：若调用失败不影响其他逻辑
                }

                // 4) 调用原始 UpdateTrajectory（仅用于远程武器）
                try
                {
                    // 攻城武器为 null，因为当前分支处理的是手持远程武器
                    ProjectileTrajectorySystem.UpdateTrajectory(player, null);
                }
                catch
                {
                    // ignore
                }
            }

            // --- 分支2: 攻城武器逻辑 ---
            if (  !isUsingRangedWeapon) // 使用非远程武器（可能是近战，也可能是操作攻城武器）
            {
                // 检查玩家是否正在使用攻城武器
                WeakGameEntity wge = player.CurrentlyUsedGameObject?.GameEntity ?? player.GetSteppedEntity();
                RangedSiegeWeapon siege = null;

                // 查找父级实体中是否有 RangedSiegeWeapon 脚本
                while (wge != null && !wge.HasScriptOfType<RangedSiegeWeapon>())
                    wge = wge.Parent;

                if (wge != null)
                    siege = wge.GetFirstScriptOfType<RangedSiegeWeapon>();

                // 5) 如果找到了攻城武器
                if (siege != null)
                {
                    bool siegeLookingUp = false;
                    try
                    {
                        // 直接使用引擎提供的方向（你所要求的）
                        Vec3 siegeDir = ProjectileTrajectorySystem.GetSiegeShootingDirection(siege);
                        siegeLookingUp = siegeDir.z > LOOK_UP_THRESHOLD;
                    }
                    catch
                    {
                        siegeLookingUp = false;
                    }

                    // 6) 为攻城武器设置目标 alpha（抬头时虚化，仅在第一人称时生效，根据设置决定）
                    if (enableLookUpBlur)
                    {
                        SetSiegeTargetAlpha(siege, siegeLookingUp ? BLUR_ALPHA : DEFAULT_ALPHA);
                    }

                    // 7) 调用原始 UpdateTrajectory（仅用于攻城武器）
                    try
                    {
                        ProjectileTrajectorySystem.UpdateTrajectory(player, siege);
                    }
                    catch
                    {
                        // ignore
                    }
                }
            }
        }

                if (agent.Team == null || agent.Team.TeamIndex == -1 || player.Team == null || !agent.Team.IsEnemyOf(player.Team)) continue;
                // 检查是否使用远程武器
                MissionWeapon weapon = agent.WieldedWeapon;
                if (weapon.IsEmpty || !weapon.CurrentUsageItem.IsRangedWeapon) continue;
                // 检查是否瞄准玩家UpdateTrajectory 
                Agent target = agent.GetTargetAgent();
                if (target == null || target != player) continue;
                // 检查是否处于攻击动作（瞄准/拉弓阶段）
                if (agent.GetCurrentActionStage(1) == Agent.ActionStage.AttackReady)
                {
                    aimingEnemies.Add(agent);
                }
            }

            // 为符合条件的敌人添加描边（红色）
            uint? targetColor = new Color(1f, 0f, 0f, 1f).ToUnsignedInteger();
            foreach (var enemy in aimingEnemies)
            {
                ApplyOutline(enemy, targetColor);
            }

            // 清理不在瞄准列表中的敌人的描边和缓存
            var toRemove = new List<Agent>();
            foreach (var kvp in _contourCache)
            {
                if (!aimingEnemies.Contains(kvp.Key))
                {
                    // 移除描边
                    kvp.Key.AgentVisuals.SetContourColor(null, true);
                    toRemove.Add(kvp.Key);
                }
            }
            foreach (var key in toRemove)
            {
                _contourCache.Remove(key);
            }
        }

        private void ApplyOutline(Agent agent, uint? color)
        {
            // 通过缓存避免重复设置相同颜色（性能优化）
            if (_contourCache.TryGetValue(agent, out var currentColor))
            {
                if (currentColor == color) return;
            }
            // 设置描边颜色
            agent.AgentVisuals.SetContourColor(color, true);
            _contourCache[agent] = color;

            // 如果敌人骑马，同时给马添加描边
            if (agent.MountAgent != null && agent.MountAgent.AgentVisuals != null)
            {
                if (_contourCache.TryGetValue(agent.MountAgent, out var mountColor))
                {
                    if (mountColor != color)
                    {
                        agent.MountAgent.AgentVisuals.SetContourColor(color, true);
                        _contourCache[agent.MountAgent] = color;
                    }
                }
                else
                {
                    agent.MountAgent.AgentVisuals.SetContourColor(color, true);
                    _contourCache[agent.MountAgent] = color;
                }
            }
        }

        // 新增方法：更新移动目标预瞄
        private void UpdateLeadPrediction(Agent player)
        {   
            // 添加开关检查
            if (!ProjectileTrajectorySettingsManager.Settings.EnableLeadPrediction)
            {
                ClearLeadPredictionLine();
                return;
            }
            if (player == null || !player.IsActive()) return;

            // 1. 获取玩家武器
            EquipmentIndex weaponIndex = player.GetPrimaryWieldedItemIndex();
            if (weaponIndex == EquipmentIndex.None)
            {
                // 没有武器时，清理预测线
                ClearLeadPredictionLine();
                return;
            }

            MissionWeapon weapon = player.Equipment[weaponIndex];
            if (weapon.IsEmpty || !weapon.CurrentUsageItem.IsRangedWeapon)
            {
                ClearLeadPredictionLine();
                return;
            }

            // 2. 选择目标（这里选择准星下的敌人，或者最近的敌人）
            Agent targetEnemy = GetNearestEnemy(player);
            if (targetEnemy == null)
            {
                ClearLeadPredictionLine();
                return;
            }

            // 3. 获取弹速
            float projectileSpeed = (float)weapon.GetModifiedMissileSpeedForCurrentUsage() *
                                   player.AgentDrivenProperties.MissileSpeedMultiplier;

            // 4. 核心计算：计算预测位置 (使用你提供的算法)
            Vec3 predictedPosition = ProjectileTrajectorySystem.CalculateLeadPosition(player, targetEnemy, projectileSpeed);

            if (predictedPosition == Vec3.Zero && predictedPosition == Vec3.Invalid)
            {
                ClearLeadPredictionLine();
                return;
            }

            // 5. 绘制辅助标记（石头）
            DrawPredictionMarker(predictedPosition);

            // 6. 绘制辅助弹道线（关键：这条线是从玩家位置指向 predictedPosition）
            ProjectileTrajectorySystem.DrawLeadTrajectory(player, predictedPosition, projectileSpeed);
        }

        // 简单的辅助函数：获取最近的敌人
        private Agent GetNearestEnemy(Agent player)
        {
            Agent nearest = null;
            float nearestDist = float.MaxValue;

            foreach (Agent agent in Mission.Current.Agents)
            {
                if (agent == player || !agent.IsHuman || agent.IsFriendOf(Agent.Main)) continue;

                float dist = (agent.Position - player.Position).LengthSquared;
                if (dist < nearestDist)
                {
                    nearestDist = dist;
                    nearest = agent;
                }
            }
            return nearest;
        }

        // 绘制预测点的标记（石头）

            if (PredictionMarker == null )

                try
                {
                    PredictionMarker = GameEntity.Instantiate(Mission.Current.Scene, PREDICTION_MESH, true);
                }
                catch { /* 如果模型找不到，忽略 */ }
            }

            if (PredictionMarker != null )
            {
                PredictionMarker.SetLocalPosition(pos);
                // 设置大小
                // PredictionMarker.SetScale(...);
            }
        }

        {

            }
        }

        internal static GameEntity GetOrCreateGameEntity()
        {
            if (_gameEntityPool.Count > 0)
            {
                var entity = _gameEntityPool.Pop();
                // 重置实体状态（例如，隐藏位置，确保它是干净的）
                entity.SetLocalPosition(new Vec3(0, 0, 0)); // 移动到原点表示已激活
                // 清除颜色缓存中对该实体的记录
                GameEntityColorCache.Remove(entity);
                return entity;
            }
            else
            {
                // 如果池为空，创建新实体
                return GameEntity.CreateEmpty(Mission.Current.Scene);
            }
        }

        internal static void ReturnGameEntityToPool(GameEntity entity)
        {
            if (entity != null)
            {
                // 将实体移回原点，表示闲置
                entity.SetLocalPosition(new Vec3(0, 0, 0));
                // 清除颜色缓存中对该实体的记录
                GameEntityColorCache.Remove(entity);
                _gameEntityPool.Push(entity);
            }
        }

        // 新增方法：更新敌人运动数据
        private void UpdateEnemyMotionData(float dt)
        {
            Agent player = Agent.Main;
            if (player == null) return;

            // 遍历所有代理，更新速度缓存
            foreach (Agent agent in Mission.Current.Agents)
            {
                if (agent == player || !agent.IsActive() || !agent.IsHuman) continue;

                Vec3 currentPos = agent.Position;
                float currentTime = Mission.Current.CurrentTime; // 使用引擎时间

                if (!TrackedAgents.ContainsKey(agent))
                {
                    TrackedAgents[agent] = new AgentMotionData
                    {
                        LastPosition = currentPos,
                        LastUpdateTime = currentTime
                    };
                    continue;
                }

                AgentMotionData data = TrackedAgents[agent];
                float timeDelta = currentTime - data.LastUpdateTime;

                // 计算速度 (防除零)
                if (timeDelta > 0.01f)
                {
                    data.Velocity = (currentPos - data.LastPosition) / timeDelta;
                    data.LastUpdateTime = currentTime;
                    data.LastPosition = currentPos;
                }
            }
        }
        private void HandleSlowMotion()
        {
            var settings = ProjectileTrajectorySettingsManager.Settings;
            if (!settings.EnableSlowMotion)
            {
                RestoreTime();
                return;
            }

            Agent player = Agent.Main;
            if (player == null || !player.IsActive())
            {
                RestoreTime();
                return;
            }

            // 1️⃣ 是否远程武器
            var weapon = player.WieldedWeapon;
            bool isRanged = !weapon.IsEmpty && weapon.CurrentUsageItem.IsRangedWeapon;

            if (!isRanged)
            {
                RestoreTime();
                return;
            }

            // 2️⃣ 是否在准备攻击
            bool isAiming = player.GetCurrentActionStage(1) == Agent.ActionStage.AttackReady;

            if (!isAiming)
            {
                RestoreTime();
                return;
            }

            // 3️⃣ 是否有敌人
            if (!HasNearbyEnemy(player, settings.SlowMoEnemyRadius))
            {
                RestoreTime();
                return;
            }

            // 触发慢动作
            ApplyTimeScale(settings.SlowMoTimeScale);
        }
        private bool HasNearbyEnemy(Agent player, float radius)
        {
            var enemies = new MBList<Agent>();

            Mission.Current.GetNearbyEnemyAgents(
                player.Position.AsVec2,
                radius,
                player.Team,
                enemies);

            return enemies.Count > 0;
        }
        // 正确做法：通过AddTimeSpeedRequest/RemoveTimeSpeedRequest管理请求
        private void ApplyTimeScale(float scale)
        {
            scale = TaleWorlds.Library.MathF.Clamp(scale, 0.05f, 1f);

            if (_slowMoActive && TaleWorlds.Library.MathF.Abs(_currentTimeScale - scale) < 0.001f)
                return;

            var mission = Mission.Current;
            if (mission != null)
            {
                mission.AddTimeSpeedRequest(new Mission.TimeSpeedRequest(scale, 999));
            }

            _slowMoActive = true;
            _currentTimeScale = scale;
        }

        private void RestoreTime()
        {
            // 1. 状态检查
            if (!_slowMoActive)
                return;

            // 2. 恢复时间流速：通过移除请求实现
            var mission = Mission.Current;
            if (mission != null)
            {
                // 移除我们添加的特定请求
                mission.RemoveTimeSpeedRequest(999);
            }

            // 3. 更新状态机
            _slowMoActive = false;
            _currentTimeScale = 1f;
        }
